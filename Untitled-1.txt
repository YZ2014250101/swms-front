swms-front/

│   └── icons.svg                    #   网站公共图标

│   ├── api/                         # 【接口请求层】集中管理所有与后端通信的请求
│   │   ├── modules/                 #   按业务端/模块拆分的接口文件目录
│   │   │   ├── student/             #   学生端所有接口（查询学生相关信息用）
│   │   │   │   ├── info.js          #     学生个人信息接口（查询/修改个人信息）
│   │   │   │   ├── schedule.js      #     学生课表接口（按周次查询课表）
│   │   │   │   ├── leave.js         #     请假申请接口（提交请假、查询请假记录）
│   │   │   │   └── score.js         #     成绩查询接口（查询各科成绩）
│   │   │   ├── teacher/             #   教师端所有接口（教师业务相关请求）
│   │   │   │   ├── course.js        #     课程管理接口（查询所授课程）
│   │   │   │   ├── student.js       #     学生管理接口（查询课程学生名单）
│   │   │   │   └── approval.js      #     审批管理接口（审批学生请假等申请）
│   │   │   └── admin/               #   管理员端所有接口（系统管理类请求）
│   │   │       ├── user.js          #     用户管理接口（增删改查用户账号）
│   │   │       ├── role.js          #     角色权限接口（角色分配、权限配置）
│   │   │       └── system.js        #     系统配置接口（系统参数、字典维护）
│   │   └── request.js               #   axios 实例封装（统一 baseURL、请求/响应拦截器、token 注入、错误处理）
│   │
│   ├── assets/                      # 【静态资源目录】存放参与构建打包的资源，会被 Vite 压缩处理
│   │   ├── images/                  #   图片资源目录
│   │   │   ├── icons/               #   业务小图标（如菜单图标、状态图标）
│   │   │   └── login/               #   登录页专用图片（如背景图、Logo）
│   │   ├── styles/                  #   全局样式文件目录
│   │   │   ├── variables.scss       #     主题变量（统一颜色、间距、字号，方便换肤）
│   │   │   ├── reset.css            #     样式重置（清除浏览器默认样式）
│   │   │   └── common.scss          #     通用样式类（如 .flex、.mt-10 等高频复用类）
│   │   └── fonts/                   #   字体文件（如自定义图标字体、特殊中文字体）
│   │
│   ├── components/                  # 【全局通用组件目录】三端共用的 Vue 组件，避免重复造轮子
│   │   ├── base/                    #   基础原子组件（最底层、无业务逻辑、可复用性最高）
│   │   │   ├── BaseButton.vue       #     统一按钮（封装样式与点击态）
│   │   │   ├── BaseInput.vue        #     统一输入框（封装校验、清空等）
│   │   │   ├── BaseTable.vue        #     统一表格（封装表头、分页占位）
│   │   │   └── BaseDialog.vue       #     统一弹窗（封装开关与遮罩）
│   │   ├── layout/                  #   布局组件（页面骨架结构，三端复用）
│   │   │   ├── AppHeader.vue        #     顶部导航栏（含用户信息、退出）
│   │   │   ├── AppSidebar.vue       #     侧边菜单栏（按角色动态渲染菜单）
│   │   │   └── AppFooter.vue        #     底部信息栏
│   │   └── business/                #   业务通用组件（含一定业务语义、跨页面复用）
│   │       ├── UserAvatar.vue       #     用户头像组件（含昵称展示）
│   │       ├── PagePagination.vue   #     分页组件（列表页通用）
│   │       └── SearchBar.vue        #     搜索栏组件（含关键词输入与查询按钮）
│   │
│   ├── views/                       # 【页面视图目录】按端划分，存放各端的页面级 .vue 文件（每个文件对应一个路由页面）
│   │   ├── login/                   #   登录注册页（三端共用统一登录入口）
│   │   │   └── Login.vue            #     登录页（账号密码登录、角色判断）
│   │   ├── student/                 #   学生端页面（仅学生可见/可访问）
│   │   │   ├── MyInfo.vue           #     个人信息页（查看/编辑学生资料）
│   │   │   ├── MySchedule.vue       #     我的课表页（按周查看课程安排）
│   │   │   ├── LeaveApply.vue       #     请假申请页（提交请假、查看审批进度）
│   │   │   └── ScoreQuery.vue       #     成绩查询页（查询各学期成绩）
│   │   ├── teacher/                 #   教师端页面（仅教师可见/可访问）
│   │   │   ├── CourseList.vue       #     我的课程页（查看所授课程列表）
│   │   │   ├── StudentManage.vue    #     学生管理页（查看/管理选课学生）
│   │   │   └── ApprovalList.vue     #     审批列表页（审批学生请假等申请）
│   │   └── admin/                   #   管理员端页面（仅管理员可见/可访问）
│   │       ├── UserManage.vue       #     用户管理页（增删改查所有用户账号）
│   │       ├── RoleManage.vue       #     角色权限页（角色与权限分配）
│   │       └── SystemConfig.vue     #     系统配置页（系统参数、字典维护）
│   │
│   ├── router/                      # 【路由目录】管理前端路由表与路由守卫
│   │   ├── index.js                 #   路由入口（创建 router 实例、合并各端路由模块）
│   │   ├── guards.js                #   路由守卫（登录校验、按角色拦截越权访问、跳转 403）
│   │   └── modules/                 #   路由按端拆分（便于维护与权限控制）
│   │       ├── student.js           #     学生端路由表
│   │       ├── teacher.js           #     教师端路由表
│   │       └── admin.js             #     管理员端路由表
│   │
│   ├── store/                       # 【状态管理目录】集中存放跨组件共享的全局状态（建议用 Pinia）
│   │   ├── index.js                 #   状态管理入口（统一导出各 store）
│   │   └── modules/                 #   按模块拆分的状态
│   │       ├── user.js              #     用户状态（登录态、token、当前角色、用户信息）
│   │       └── app.js               #     应用 UI 状态（侧边栏折叠、主题等全局界面状态）
│   │
│   ├── utils/                       # 【工具函数目录】存放纯逻辑工具函数，与业务无关、可随处复用
│   │   ├── auth.js                  #   token 存取（封装 localStorage 的 get/set/remove）
│   │   ├── format.js                #   格式化工具（日期格式化、金额格式化等）
│   │   ├── validate.js              #   表单校验规则（手机号、邮箱、学号等校验正则）
│   │   └── permission.js            #   权限判断函数（判断当前用户是否有某按钮权限）
│   │
│   ├── constants/                   # 【常量定义目录】存放全局不变的枚举值与配置，避免代码中出现魔法字符串
│   │   ├── role.js                  #   角色枚举（STUDENT / TEACHER / ADMIN）
│   │   └── menu.js                  #   各端菜单配置（学生/教师/管理员各自的菜单项）
│   │
│   ├── directives/                  # 【自定义指令目录】存放 Vue 自定义指令
│   │   └── permission.js            #   v-permission 指令（按权限控制按钮/元素的显示隐藏）
│   │
│   ├── hooks/                       # 【组合式函数目录】存放可复用的 Vue3 Composition API 逻辑
│   │   ├── useTable.js              #   表格逻辑复用（封装分页、加载、查询的通用逻辑）
│   │   └── useDialog.js             #   弹窗逻辑复用（封装弹窗开关状态的通用逻辑）
│   │
│   ├── App.vue                      # 【根组件】整个应用的入口组件，承载路由出口
│   ├── main.js                      # 【入口文件】应用启动入口（挂载 Vue、注册路由与状态管理）
│   └── style.css                    # 【全局基础样式】最基础的全局样式
│
├── .env.development                 # 【开发环境变量】开发环境专用配置（如开发接口地址 VITE_API_BASE_URL）
├── .env.production                  # 【生产环境变量】生产环境专用配置（如线上接口地址）
├── .gitignore                       # Git 忽略规则（指定哪些文件/目录不纳入版本管理）
├── index.html                       # 【HTML 模板】Vite 入口 HTML，挂载 Vue 应用的根节点
├── vite.config.js                   # 【Vite 配置】构建配置（路径别名 @、开发代理 proxy、插件等）
└── package.json                     # 【项目配置】依赖声明与 npm 脚本（dev/build/preview）
